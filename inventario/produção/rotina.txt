import sys

    session = boto3.Session(profile_name=profile_name)
    
    # Cria um cliente para o serviço EMR
    emr_client = session.client('emr')
    
    # Lista os clusters EMR
    response = emr_client.list_clusters(ClusterStates=['STARTING', 'BOOTSTRAPPING', 'RUNNING', 'WAITING', 'TERMINATING'])
    
    # Para cada cluster, obter as tags e imprimir a tag "Name"
    for cluster in response['Clusters']:
        cluster_id = cluster['Id']
        tags_response = emr_client.list_tags(ResourceId=cluster_id)
        
        # Procura pela tag "Name" nas tags do cluster
        for tag in tags_response['Tags']:
            if tag['Key'] == 'Name':
                print(f"Cluster ID: {cluster_id}, Name: {tag['Value']}")

# Aqui está uma abordagem corrigida e mais genérica para tentar abordar múltiplos valores de tags em uma consulta, focando em estruturar corretamente o filtro dentro das limitações explicadas:

import boto3
from datetime import datetime

# Configuração inicial
aws_profile = "meu_perfil_aws"  # Substitua pelo seu perfil AWS
tags_names = ["ClusterName1", "ClusterName2"]  # Substitua pelos nomes dos seus clusters EMR

# Configura a sessão utilizando o perfil AWS fornecido
session = boto3.Session(profile_name=aws_profile)

# Cria um cliente para o AWS Cost Explorer
ce_client = session.client('ce')

# Define o período de tempo para o mês corrente
now = datetime.utcnow()
start_date = f"{now.year}-{now.month:02d}-01"
end_date = f"{now.year}-{now.month:02d}-{now.day:02d}"

# Ajuste na abordagem para múltiplos valores de tag
# Devido às limitações mencionadas, a filtragem por múltiplos valores de tags pode necessitar consultas individuais ou análise posterior

# Executa uma consulta de custo e uso para cada tag "Name"
for tag_name in tags_names:
    response = ce_client.get_cost_and_usage(
        TimePeriod={'Start': start_date, 'End': end_date},
        Granularity='MONTHLY',
        Metrics=["UnblendedCost"],
        Filter={
            "Tags": {
                "Key": "Name",
                "Values": [tag_name]
            }
        },
        GroupBy=[
            {"Type": "DIMENSION", "Key": "SERVICE"}
        ]
    )

    # Processa e imprime os resultados para cada tag
    print(f"Results for tag 'Name': {tag_name}")
    for result in response['ResultsByTime']:
        for group in result['Groups']:
            service = group['Keys'][0]
            amount = group['Metrics']['UnblendedCost']['Amount']
            print(f"Service: {service}, Unblended Cost: ${amount}")
    print("\n")

# Neste script, a consulta get_cost_and_usage é chamada separadamente para cada valor de tag_name em tags_names. Isso significa que para cada tag de nome de cluster EMR, uma consulta separada é realizada, e os custos são recuperados e impressos para aquela tag específica.

# Essa abordagem pode não ser a mais eficiente em termos de desempenho, especialmente se você tiver muitas tags para consultar, mas contorna a limitação de aplicar múltiplos critérios de filtro em uma única chamada de API. Para análises mais complexas ou se você estiver lidando com um grande volume de tags, considerar a exportação dos dados de custo para análise offline ou o uso de ferramentas de análise de dados pode ser mais adequado.
