## Parameter store

## Dynamodb table 
# obter valores de DEV
aws dynamodb describe-table \
  --profile datahubdev \
  --region sa-east-1 \
  --table-name tb_positivo_mercantil_controle_remessa

# criar em STAGE
aws dynamodb create-table \
  --profile datahubstage \
  --region sa-east-1 \
  --table-name tb_positivo_mercantil_controle_remessa \
  --attribute-definitions \
      AttributeName=fonte_remessa,AttributeType=S \
      AttributeName=nu_remessa,AttributeType=N \
  --key-schema \
      AttributeName=fonte_remessa,KeyType=HASH \
      AttributeName=nu_remessa,KeyType=RANGE \
  --billing-mode PAY_PER_REQUEST
